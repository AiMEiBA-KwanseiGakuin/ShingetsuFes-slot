import sys
from PySide6.QtWidgets import QApplication, QWidget, QPushButton, QLabel
from PySide6.QtCore import Qt


# =========================
# キャンバス
# =========================

class Canvas(QWidget):

    def __init__(self):
        super().__init__()

        # キャンバス直属のカード
        self.cards = []

    def add_card(self, card):
        if card not in self.cards:
            self.cards.append(card)

    def remove_card(self, card):
        if card in self.cards:
            self.cards.remove(card)


# =========================
# カード
# =========================

class Card(QLabel):

    def __init__(self, parent=None, parent_card=None):
        super().__init__("カード", parent)

        # ロイロ上での親カード
        self.parent_card = parent_card

        # このカードの子カード
        self.children_cards = []

        # ドラッグ用
        self.dragging = False
        self.offset = None

        self.setStyleSheet("""
            background-color: white;
            color: black;
            border: 1px solid gray;
            padding: 10px;
        """)

        self.resize(200, 100)

    # =========================
    # マウスを押す
    # =========================

    def mousePressEvent(self, event):

        if event.button() == Qt.LeftButton:

            self.dragging = True
            self.offset = event.position().toPoint()

    # =========================
    # ドラッグ
    # =========================

    def mouseMoveEvent(self, event):

        if self.dragging:

            new_position = (
                self.pos()
                + event.position().toPoint()
                - self.offset
            )

            self.move(new_position)

    # =========================
    # マウスを離す
    # =========================

    def mouseReleaseEvent(self, event):

        if event.button() == Qt.LeftButton:

            self.dragging = False

            # 今いる場所がキャンバス上か確認
            if self.parent_card is None:

                target = self.find_target_card()

                # 別のカードの上で離した
                if target is not None and target != self:

                    self.put_inside(target)

    # =========================
    # 入れる相手を探す
    # =========================

    def find_target_card(self):

        # キャンバス上のカードを調べる
        for card in canvas.cards:

            if card == self:
                continue

            # 自分の中心点
            center = self.geometry().center()

            # キャンバス座標に変換
            center_on_canvas = self.mapTo(canvas, center)

            # 相手の範囲に入っているか
            target_rect = card.geometry()

            if target_rect.contains(center_on_canvas):
                return card

        return None

    # =========================
    # 別のカードの中に入る
    # =========================

    def put_inside(self, target):

        # キャンバスから削除
        canvas.remove_card(self)

        # 新しい親カードを設定
        self.parent_card = target

        # 親カードの子リストに追加
        target.children_cards.append(self)

        # Qt上の親も変更
        self.setParent(target)

        # 親カードの中の位置
        self.move(20, 20)

        self.show()

    # =========================
    # ダブルクリックで開く
    # =========================

    def mouseDoubleClickEvent(self, event):

        if event.button() == Qt.LeftButton:

            self.editor = CardEditor(self)
            self.editor.show()


# =========================
# カードを開いた画面
# =========================

class CardEditor(QWidget):

    def __init__(self, card):
        super().__init__()

        self.card = card

        self.setWindowTitle("カード")
        self.resize(700, 500)

        self.setStyleSheet("""
            background-color: darkblue;
        """)

        # =========================
        # ツールバー
        # =========================

        self.add_button = QPushButton(
            "＋カード",
            self
        )
        self.add_button.move(20, 20)

        self.color_button = QPushButton(
            "色変更",
            self
        )
        self.color_button.move(120, 20)

        self.text_button = QPushButton(
            "文字編集",
            self
        )
        self.text_button.move(220, 20)

        # =========================
        # 大きなカード
        # =========================

        self.card_view = QLabel(
            self.card.text(),
            self
        )

        self.card_view.setStyleSheet("""
            background-color: white;
            color: black;
            border: 1px solid gray;
            padding: 20px;
        """)

        self.card_view.resize(600, 350)
        self.card_view.move(50, 90)

        # =========================
        # ＋カード
        # =========================

        self.add_button.clicked.connect(
            self.create_child_card
        )

    # =========================
    # 子カードを作る
    # =========================

    def create_child_card(self):

        child = Card(
            self,
            parent_card=self.card
        )

        # 親カードに登録
        self.card.children_cards.append(child)

        number = len(self.card.children_cards)

        x = 30 + (number - 1) * 20
        y = 100 + (number - 1) * 20

        child.move(x, y)

        child.show()


# =========================
# アプリ
# =========================

app = QApplication(sys.argv)


# =========================
# キャンバス
# =========================

canvas = Canvas()

canvas.setWindowTitle("My Note")
canvas.resize(800, 600)

canvas.setStyleSheet("""
    background-color: darkblue;
""")


# =========================
# カードを作る
# =========================

def create_card():

    card = Card(canvas)

    x = 100 + len(canvas.cards) * 20
    y = 100 + len(canvas.cards) * 20

    card.move(x, y)

    card.show()

    canvas.add_card(card)


# =========================
# カード追加ボタン
# =========================

button = QPushButton(
    "カードを作る",
    canvas
)

button.move(20, 20)

button.clicked.connect(create_card)


# =========================
# 表示
# =========================

canvas.show()

sys.exit(app.exec())
